import cv2  
import requests  
import json
import base64

def test_detect_face():
    # 读取图像  
    image = cv2.imread('./test_images/mahuateng_2.jpg')  
    # 将图像转换为Base64编码的字符串
    _, img_encoded = cv2.imencode('.jpg', image)
    image_base64 = base64.b64encode(img_encoded).decode('utf-8')

    # 构造JSON请求体
    data = {
        'image': image_base64,
        "image_type":"BASE64"
    }
    json_data = json.dumps(data)

    # 发送HTTP请求
    url = 'http://10.178.11.151:8003/detect'
    headers = {'Content-Type': 'application/json'}

    response = requests.request("POST", url, headers=headers, data=json_data,stream=False)
    # 检查响应状态码
    if response.status_code == 200:
        print('图像成功发送到服务器！')
        data = json.loads(response.content)
        print(data)
    else:
        print('发送图像时出现问题。')
